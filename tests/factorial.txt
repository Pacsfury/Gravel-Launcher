; ModuleID = 'output.ll'
declare i32 @putchar(i32)


define void @cprint(i32 %charasc) {
entry:
    %0 = call i32 @putchar(i32 %charasc)
    ret void
}


define i32 @factorial(i32 %n) {
entry:
    %n.addr = alloca i32, align 4
    store i32 %n, ptr %n.addr, align 4
    %1 = load i32, ptr %n.addr, align 4
    %2 = icmp eq i32 %1, 1
    %3 = zext i1 %2 to i32
    %4 = icmp ne i32 %3, 0
    br i1 %4, label %then1, label %else2
then1:
    ret i32 1
else2:
    br label %end3
end3:
    %5 = load i32, ptr %n.addr, align 4
    %6 = load i32, ptr %n.addr, align 4
    %7 = sub i32 %6, 1
    %8 = call i32 @factorial(i32 %7)
    %9 = mul i32 %5, %8
    ret i32 %9
}

define i32 @main() {
entry:
    %1 = call i32 @factorial(i32 5)
    %2 = icmp eq i32 %1, 120
    %3 = zext i1 %2 to i32
    %4 = icmp ne i32 %3, 0
    br i1 %4, label %then4, label %else5
then4:
    call void @cprint(i32 97)
    br label %end6
else5:
    br label %end6
end6:
    ret i32 0
}
