declare i32 @putchar(i32)

@x = global i32 0, align 4
@addr = global ptr null, align 4
@y = global i32 0, align 4

define void @cprint(i32 %charasc) {
entry:
    %0 = call i32 @putchar(i32 %charasc)
    ret void
}


define i32 @main() {
entry:
    store i32 42, ptr @x, align 4
    %1 = getelementptr inbounds i32, ptr @x, i32 0
    store ptr %1, ptr @addr, align 4
    %2 = load ptr, ptr @addr, align 4
    %3 = load i32, ptr %2, align 4
    store i32 %3, ptr @y, align 4
    %4 = load i32, ptr @y, align 4
    call void @cprint(i32 %4)
    ret i32 0
}
