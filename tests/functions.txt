declare i32 @putchar(i32)


define void @cprint(i32 %charasc) {
entry:
    %0 = call i32 @putchar(i32 %charasc)
    ret void
}


define i32 @voidFun() {
entry:
    call void @cprint(i32 0)
    ret i32 0
}

define i32 @intFun() {
entry:
    ret i32 65
}

define i32 @plusFive(i32 %n) {
entry:
    %n.addr = alloca i32, align 4
    store i32 %n, ptr %n.addr, align 4
    %1 = load i32, ptr %n.addr, align 4
    %2 = add i32 %1, 5
    ret i32 %2
}

define i32 @main() {
entry:
    %1 = call i32 @intFun()
    call void @cprint(i32 %1)
    %2 = call i32 @plusFive(i32 63)
    call void @cprint(i32 %2)

}
